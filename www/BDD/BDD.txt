-- Table des rôles système
CREATE TABLE roles (
  id  INT AUTO_INCREMENT PRIMARY KEY,
  nom VARCHAR(50) NOT NULL UNIQUE
);
INSERT INTO roles (nom) VALUES ('admin'), ('employe'), ('utilisateur');

-- Table des utilisateurs
CREATE TABLE utilisateurs (
  id       INT AUTO_INCREMENT PRIMARY KEY,
  nom      VARCHAR(100) NOT NULL,
  email    VARCHAR(100) NOT NULL UNIQUE,
  password VARCHAR(255) NOT NULL,
  credits  INT          DEFAULT 20,
  role_id  INT          DEFAULT 3,
  actif    BOOLEAN      DEFAULT TRUE,
  photo    VARCHAR(255) DEFAULT NULL,
  FOREIGN KEY (role_id) REFERENCES roles(id)
);

-- Rôle métier (passager / chauffeur / les deux)
CREATE TABLE roles_utilisateurs (
  id              INT AUTO_INCREMENT PRIMARY KEY,
  utilisateur_id  INT NOT NULL,
  role            ENUM('passager','chauffeur','chauffeur_passager') NOT NULL,
  date_attribution DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (utilisateur_id) REFERENCES utilisateurs(id) ON DELETE CASCADE
);

-- Véhicules
CREATE TABLE vehicules (
  id            INT AUTO_INCREMENT PRIMARY KEY,
  utilisateur_id INT NOT NULL,
  plaque        VARCHAR(20)  NOT NULL,
  date_immat    DATE         NOT NULL,
  modele        VARCHAR(100) NOT NULL,
  marque        VARCHAR(100) NOT NULL,
  couleur       VARCHAR(50)  NOT NULL,
  places        INT          NOT NULL,
  preferences   TEXT,
  type_vehicule ENUM('essence','diesel','electrique','hybride') NOT NULL DEFAULT 'essence',
  date_ajout    TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (utilisateur_id) REFERENCES utilisateurs(id) ON DELETE CASCADE
);

-- Covoiturages
CREATE TABLE covoiturage (
  covoiturage_id INT AUTO_INCREMENT PRIMARY KEY,
  utilisateur_id INT          NOT NULL,
  vehicule_id    INT,
  lieu_depart    VARCHAR(100) NOT NULL,
  lieu_arrivee   VARCHAR(100) NOT NULL,
  date_depart    DATE         NOT NULL,
  heure_depart   TIME         NOT NULL,
  heure_arrivee  TIME         NULL,
  nb_place       INT          NOT NULL,
  prix_personne  FLOAT        NOT NULL,
  type_vehicule  ENUM('essence','diesel','electrique','hybride'),
  statut         ENUM('disponible','complet','en_cours','termine','annule') DEFAULT 'disponible',
  FOREIGN KEY (utilisateur_id) REFERENCES utilisateurs(id),
  FOREIGN KEY (vehicule_id)    REFERENCES vehicules(id) ON DELETE SET NULL
);

-- Participations
CREATE TABLE participation (
  covoiturage_id     INT NOT NULL,
  utilisateur_id     INT NOT NULL,
  statut             ENUM('en_attente','validee','litige') DEFAULT 'en_attente',
  commentaire_litige TEXT,
  PRIMARY KEY (covoiturage_id, utilisateur_id),
  FOREIGN KEY (covoiturage_id) REFERENCES covoiturage(covoiturage_id) ON DELETE CASCADE,
  FOREIGN KEY (utilisateur_id) REFERENCES utilisateurs(id) ON DELETE CASCADE
);

-- Avis passagers
CREATE TABLE avis (
  avis_id        INT AUTO_INCREMENT PRIMARY KEY,
  covoiturage_id INT          NOT NULL,
  utilisateur_id INT,
  auteur         VARCHAR(100) DEFAULT 'Anonyme',
  commentaire    VARCHAR(255),
  note           INT          CHECK (note >= 1 AND note <= 5),
  statut         ENUM('publié','en_attente','rejeté') DEFAULT 'en_attente',
  date_avis      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (covoiturage_id) REFERENCES covoiturage(covoiturage_id) ON DELETE CASCADE,
  FOREIGN KEY (utilisateur_id) REFERENCES utilisateurs(id) ON DELETE SET NULL
);
